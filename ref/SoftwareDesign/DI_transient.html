純粋な依存注入での生存戦略の選択

図8.4　オブジェクト・グラフ

CommerceControllerActivator クラスによって生成されるオブジェクト・グラフ。 HomeController クラスと RouteController クラスのオブジェクトが必要な依存と共に生成される。

シングルトン（使い回す） 短命（リクエストごとに新規生成）

リスト8.3　CommerceControllerActivator

コンストラクタでシングルトン対象を一度だけ生成して readonly フィールドに格納し、 Create が呼ばれるたびに短命な依存を新規生成して注入する。 

public class CommerceControllerActivator : IControllerActivator
{
    private readonly RouteCalculator calculator;
    private readonly AspNetUserContextAdapter userContext;
    private readonly string connectionString;

    public CommerceControllerActivator(string connectionString)
    {
        // シングルトン対象は一度だけ生成して保持する
        this.calculator = new RouteCalculator();
        this.userContext = new AspNetUserContextAdapter();
        this.connectionString = connectionString;
    }

    // IControllerActivator が定める「コントローラを生成する係」のフック。
    // ASP.NET Core MVC は、コントローラを直接 new せず、登録された
    // IControllerActivator にコントローラ生成を委譲する設計になっている。
    // そのため Create は、フレームワークがリクエストを受け取り
    // コントローラを必要とするたび（＝リクエストごと）に呼び出される。
    // 渡される context から対象のコントローラ型を判別し、その型に応じた
    // オブジェクト・グラフを組み立てて返す。
    public object Create(ControllerContext context)
    {
        Type type = context.ActionDescriptor.ControllerTypeInfo.AsType();

        switch (type.Name)
        {
            case nameof(HomeController):
                return CreateHomeController();
            case nameof(RouteController):
                return CreateRouteController();
            default:
                throw new Exception("Unknown controller type.");
        }
    }

    private HomeController CreateHomeController()
    {
        // 短命な依存は Create が呼ばれるたびに新規生成する
        return new HomeController(
            new ProductService(
                new SqlProductRepository(
                    new CommerceContext(this.connectionString),
                    this.userContext)));
    }

    private RouteController CreateRouteController()
    {
        // シングルトンの calculator を受け取るだけ
        return new RouteController(this.calculator);
    }

    public void Release(ControllerContext context, object controller)
    {
    }
}
